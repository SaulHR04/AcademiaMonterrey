CREATE DATABASE db_Youtubers;
use db_Youtubers;
CREATE TABLE IF NOT EXISTS youtubers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    canal_youtube VARCHAR(100) NOT NULL,
    tipo_contenido VARCHAR(100),
    seguidores VARCHAR(50),
    monetizacion BOOLEAN DEFAULT TRUE,
    videos_total INT DEFAULT 0,
    funas INT DEFAULT 0,
    tiempo_youtube VARCHAR(50),
    equipo_de_trabajo VARCHAR(150)
);
